import java.io.IOException;
import java.io.InputStream;
import org.apache.tika.exception.TikaException;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;

class EndianUtils_readIntBE_7_0_Test {

    @Test
    void testReadIntBE() throws IOException {
        InputStream stream = mock(InputStream.class);
        when(stream.read()).thenReturn(0x12, 0x34, 0x56, 0x78);
        assertEquals(0x12345678, EndianUtils.readIntBE(stream));
    }

    @Test
    void testReadIntBEWithUnderflow() throws IOException {
        InputStream stream = mock(InputStream.class);
        when(stream.read()).thenReturn(0x12, 0x34, 0x56, -1);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readIntBE(stream));
    }
}

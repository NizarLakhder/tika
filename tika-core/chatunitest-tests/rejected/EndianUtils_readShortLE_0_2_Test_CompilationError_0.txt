import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

class EndianUtils_readShortLE_0_2_Test {

    @Test
    void testReadShortLE() throws IOException, TikaException {
        InputStream stream = new ByteArrayInputStream(new byte[] { 1, 2 });
        short result = EndianUtils.readShortLE(stream);
        assertEquals(258, result);
    }

    @Test
    void testReadShortLEWithUnderflow() throws IOException, TikaException {
        InputStream stream = new ByteArrayInputStream(new byte[] { 1 });
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readShortLE(stream));
    }

    @Test
    void testReadShortLEWithNullStream() {
        assertThrows(NullPointerException.class, () -> EndianUtils.readShortLE(null));
    }
